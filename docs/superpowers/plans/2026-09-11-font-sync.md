# Font Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Font files the document asks for travel from the Photoshop PC to the app on another device, through the relay, and are kept on that device.

**Architecture:** The app, after the layer tree arrives, sends `font-request` for the names it is substituting. The plugin finds each file by reading its sfnt `name` table in folders the user picked once, and answers with `font-chunk` frames (512 KiB raw, base64) or `font-missing`. The app reassembles, verifies SHA-256, loads through the existing `FontsService.assignTo`, and caches the bytes in IndexedDB.

**Tech Stack:** plain ES modules in `src/sync/` (vendored to CommonJS for UXP), UXP `storage.localFileSystem`, Cloudflare Durable Object relay + Node relay, Angular 22 signals, IndexedDB, vitest.

**Spec:** `docs/superpowers/specs/2026-09-11-font-sync-design.md`

## Global Constraints

- Max font file: `FONT_MAX_BYTES = 32 * 1024 * 1024`. Chunk: `FONT_CHUNK_BYTES = 512 * 1024` raw → ≤ 64 chunks.
- `font-request`: 1–64 names, each 1–128 chars. `font-missing.reason` ∈ `no-folder`, `not-found`, `collection`, `too-large`, `unreadable`.
- Direction: `font-request` app → plugin only; `font-chunk`, `font-missing` plugin → app only.
- Device cache cap: `150 * 1024 * 1024` bytes, least-recently-used evicted. Hand-picked fonts are NOT cached.
- UXP: CommonJS only, no `import`, no `innerHTML`. Plugin CSS stays plain CSS (UXP), app styling is Tailwind.

```js
const push16 = (a, v) => a.push((v >>> 8) & 255, v & 255);
const push32 = (a, v) => a.push((v >>> 24) & 255, (v >>> 16) & 255, (v >>> 8) & 255, v & 255);
const utf16be = (s) => { const out = []; for (let i = 0; i < s.length; i++) push16(out, s.charCodeAt(i)); return out; };
const ascii = (s) => Array.from(s, (c) => c.charCodeAt(0));

export function makeFont(postscriptName, {
  family = 'Test Family', subfamily = 'Regular', signature = '\0\x01\0\0',
  platform = 'windows', padding = 0, base = 0,
} = {}) {
  const strings = [];
  const records = [];
  const add = (platformID, encodingID, languageID, nameID, bytes) => {
    records.push([platformID, encodingID, languageID, nameID, bytes.length, strings.length]);
    strings.push(...bytes);
  };
  for (const [id, s] of [[1, family], [2, subfamily], [6, postscriptName]]) {
    if (platform === 'mac' || platform === 'both') add(1, 0, 0, id, ascii(platform === 'both' ? `mac-${s}` : s));
    if (platform === 'windows' || platform === 'both') add(3, 1, 0x409, id, utf16be(s));
  }
  const name = [];
  push16(name, 0); push16(name, records.length); push16(name, 6 + records.length * 12);
  for (const r of records) r.forEach((v) => push16(name, v));
  name.push(...strings);

  const head = ascii(signature);
  push16(head, 1); push16(head, 16); push16(head, 0); push16(head, 0);
  const dir = ascii('name');
  push32(dir, 0); push32(dir, base + 12 + 16); push32(dir, name.length);

  const out = new Uint8Array(head.length + dir.length + name.length + padding);
  out.set(head, 0); out.set(dir, head.length); out.set(name, head.length + dir.length);
  return out;
}

export function makeCollection(postscriptNames) {
  const header = 12 + 4 * postscriptNames.length;
  const offsets = [];
  let at = header;
  for (const n of postscriptNames) { offsets.push(at); at += makeFont(n).length; }
  const out = new Uint8Array(at);
  const h = ascii('ttcf');
  push32(h, 0x00010000); push32(h, postscriptNames.length);
  offsets.forEach((o) => push32(h, o));
  out.set(h, 0);
  postscriptNames.forEach((n, i) => out.set(makeFont(n, { base: offsets[i] }), offsets[i]));
  return out;
}
```

```js
/**
 * Reading a font file's own name, and moving its bytes over a JSON wire.
 *
 * Photoshop's API lists the fonts it has (`app.fonts`: `postScriptName`,
 * `family`, `style`) but never where their files are. So the plugin has to
 * open files and ask each one what it is. The answer is the sfnt `name`
 * table's nameID 6 -- the PostScript name -- which is the same string a PSD
 * stores. A file name is never trusted for that: `HelvNeue_Rm.otf` names
 * nothing, and `candidateOrder` below only decides which files to OPEN first.
 *
 * Pure, and imported by both ends: vendored into the plugin as CommonJS, and
 * used by the app to reassemble and verify what arrived. Nothing here throws
 * on bad input; a file that cannot be read returns null.
 */
import { FONT_CHUNK_BYTES } from './protocol.js';

const u16 = (b, o) => (b[o] << 8) | b[o + 1];
const u32 = (b, o) => ((b[o] << 24) >>> 0) + (b[o + 1] << 16) + (b[o + 2] << 8) + b[o + 3];
const tag = (b, o) => String.fromCharCode(b[o], b[o + 1], b[o + 2], b[o + 3]);

/** TrueType, Apple TrueType, and CFF-flavoured OpenType. */
const SFNT_SIGNATURES = new Set(['\0\x01\0\0', 'true', 'OTTO']);

/** Bytes from whatever the host handed over, across realms (UXP, a vm, a worker). */
function toBytes(input) {
  if (input instanceof Uint8Array) return input;
  if (ArrayBuffer.isView(input)) return new Uint8Array(input.buffer, input.byteOffset, input.byteLength);
  if (input && typeof input.byteLength === 'number') return new Uint8Array(input);
  return new Uint8Array(0);
}

const utf16 = (b, at, length) => {
  let s = '';
  for (let i = 0; i + 1 < length; i += 2) s += String.fromCharCode(u16(b, at + i));
  return s;
};
const latin = (b, at, length) => {
  let s = '';
  for (let i = 0; i < length; i++) s += String.fromCharCode(b[at + i]);
  return s;
};

/**
 * The names of the one font whose offset table starts at `start`.
 *
 * Record preference: Windows Unicode US-English, any Windows Unicode, the
 * Unicode platform, then Mac Roman. The PostScript name is ASCII by
 * definition and normally identical across records; the order only matters
 * for a file that disagrees with itself.
 */
function namesAt(b, start) {
  if (start < 0 || start + 12 > b.length || !SFNT_SIGNATURES.has(tag(b, start))) return null;
  const numTables = u16(b, start + 4);
  const dirEnd = start + 12 + numTables * 16;
  if (dirEnd > b.length) return null;

  let table = -1;
  let tableLength = 0;
  for (let rec = start + 12; rec < dirEnd; rec += 16) {
    if (tag(b, rec) === 'name') { table = u32(b, rec + 8); tableLength = u32(b, rec + 12); break; }
  }
  if (table < 0 || table + 6 > b.length || table + tableLength > b.length) return null;

  const count = u16(b, table + 2);
  const strings = table + u16(b, table + 4);
  if (table + 6 + count * 12 > b.length) return null;

  const best = new Map();
  for (let i = 0; i < count; i++) {
    const r = table + 6 + i * 12;
    const platform = u16(b, r);
    const encoding = u16(b, r + 2);
    const language = u16(b, r + 4);
    const nameID = u16(b, r + 6);
    if (nameID !== 1 && nameID !== 2 && nameID !== 6) continue;
    const length = u16(b, r + 8);
    const at = strings + u16(b, r + 10);
    if (at + length > b.length) continue;

    let rank;
    let value;
    if (platform === 3 && (encoding === 1 || encoding === 10)) { rank = language === 0x409 ? 4 : 3; value = utf16(b, at, length); }
    else if (platform === 0) { rank = 2; value = utf16(b, at, length); }
    else if (platform === 1 && encoding === 0) { rank = 1; value = latin(b, at, length); }
    else continue;

    const prev = best.get(nameID);
    if (!prev || rank > prev.rank) best.set(nameID, { rank, value });
  }

  const postscriptName = best.get(6)?.value;
  if (!postscriptName) return null;
  return { postscriptName, family: best.get(1)?.value ?? '', subfamily: best.get(2)?.value ?? '' };
}

/**
 * What font(s) a file holds.
 *
 * @returns {{postscriptName:string,family:string,subfamily:string}
 *   | {collection:true,fonts:Array} | {unsupported:'woff'} | null}
 *   A collection is reported with its members so the plugin can say "that
 *   font is inside cambria.ttc" -- which `FontFace` cannot load -- rather than
 *   the less useful "not found".
 */
export function fontNamesIn(input) {
  const b = toBytes(input);
  if (b.length < 12) return null;
  const signature = tag(b, 0);
  if (signature === 'wOFF' || signature === 'wOF2') return { unsupported: 'woff' };
  if (signature === 'ttcf') {
    const n = u32(b, 8);
    if (n < 1 || n > 256 || 12 + n * 4 > b.length) return null;
    const fonts = [];
    for (let i = 0; i < n; i++) {
      const names = namesAt(b, u32(b, 12 + i * 4));
      if (names) fonts.push(names);
    }
    return { collection: true, fonts };
  }
  return namesAt(b, 0);
}

const squash = (s) => String(s ?? '').toLowerCase().replace(/\.[a-z0-9]+$/, '').replace(/[^a-z0-9]/g, '');

/**
 * The file names, reordered so the ones resembling a wanted name open first.
 *
 * An ORDERING HINT, not a matcher: every file it puts first is still opened
 * and asked its name, so a wrong guess costs one read and never a wrong font.
 * That is why it can be this crude, and why it is not a second copy of the
 * matching in `src/fonts.js`.
 */
export function candidateOrder(names, fileNames) {
  const wants = (Array.isArray(names) ? names : [names]).map(squash).filter(Boolean);
  const score = (file) => {
    const s = squash(file);
    let best = 2;
    for (const w of wants) {
      if (s === w) return 0;
      if (s.length >= 4 && w.length >= 4 && (w.startsWith(s.slice(0, 4)) || s.startsWith(w.slice(0, 4)))) best = 1;
    }
    return best;
  };
  return fileNames
    .map((file, i) => [score(file), i, file])
    .sort((x, y) => x[0] - y[0] || x[1] - y[1])
    .map((entry) => entry[2]);
}

const B64 = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
const B64_VALUE = (() => {
  const t = new Int16Array(128).fill(-1);
  for (let i = 0; i < 64; i++) t[B64.charCodeAt(i)] = i;
  return t;
})();

/** Ours, not `btoa`: UXP's `btoa` over binary is not something to bet a font on. */
export function bytesToBase64(input) {
  const b = toBytes(input);
  let out = '';
  let i = 0;
  for (; i + 2 < b.length; i += 3) {
    const n = (b[i] << 16) | (b[i + 1] << 8) | b[i + 2];
    out += B64[n >>> 18] + B64[(n >>> 12) & 63] + B64[(n >>> 6) & 63] + B64[n & 63];
  }
  const rest = b.length - i;
  if (rest === 1) {
    const n = b[i] << 16;
    out += B64[n >>> 18] + B64[(n >>> 12) & 63] + '==';
  } else if (rest === 2) {
    const n = (b[i] << 16) | (b[i + 1] << 8);
    out += B64[n >>> 18] + B64[(n >>> 12) & 63] + B64[(n >>> 6) & 63] + '=';
  }
  return out;
}

const b64At = (s, i) => {
  const c = s.charCodeAt(i);
  return c < 128 ? B64_VALUE[c] : -1;
};

/** The bytes, or null for anything that is not well-formed base64. */
export function base64ToBytes(text) {
  if (typeof text !== 'string' || text.length % 4 !== 0) return null;
  const pad = text.endsWith('==') ? 2 : text.endsWith('=') ? 1 : 0;
  const out = new Uint8Array((text.length / 4) * 3 - pad);
  let o = 0;
  for (let i = 0; i < text.length; i += 4) {
    const last = i + 4 === text.length;
    const c0 = b64At(text, i);
    const c1 = b64At(text, i + 1);
    const c2 = last && pad === 2 ? 0 : b64At(text, i + 2);
    const c3 = last && pad >= 1 ? 0 : b64At(text, i + 3);
    if (c0 < 0 || c1 < 0 || c2 < 0 || c3 < 0) return null;
    const n = (c0 << 18) | (c1 << 12) | (c2 << 6) | c3;
    if (o < out.length) out[o++] = n >>> 16;
    if (o < out.length) out[o++] = (n >>> 8) & 255;
    if (o < out.length) out[o++] = n & 255;
  }
  return out;
}

/**
 * SHA-256, as lowercase hex.
 *
 * Ours because the two ends differ: the app has `crypto.subtle`, and UXP's
 * support for it is not documented. One implementation on both ends means
 * the checksum cannot disagree for a reason that is not the bytes.
 */
export function sha256Hex(input) {
  const bytes = toBytes(input);
  const padded = new Uint8Array(((bytes.length + 9 + 63) >> 6) << 6);
  padded.set(bytes);
  padded[bytes.length] = 0x80;
  const view = new DataView(padded.buffer);
  const bits = bytes.length * 8;
  view.setUint32(padded.length - 8, Math.floor(bits / 0x100000000));
  view.setUint32(padded.length - 4, bits >>> 0);

  const h = new Uint32Array([0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19]);
  const w = new Uint32Array(64);
  for (let off = 0; off < padded.length; off += 64) {
    for (let i = 0; i < 16; i++) w[i] = view.getUint32(off + i * 4);
    for (let i = 16; i < 64; i++) {
      const x = w[i - 15];
      const y = w[i - 2];
      const s0 = ((x >>> 7) | (x << 25)) ^ ((x >>> 18) | (x << 14)) ^ (x >>> 3);
      const s1 = ((y >>> 17) | (y << 15)) ^ ((y >>> 19) | (y << 13)) ^ (y >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
    }
    let a = h[0], b = h[1], c = h[2], d = h[3], e = h[4], f = h[5], g = h[6], k = h[7];
    for (let i = 0; i < 64; i++) {
      const S1 = ((e >>> 6) | (e << 26)) ^ ((e >>> 11) | (e << 21)) ^ ((e >>> 25) | (e << 7));
      const ch = (e & f) ^ (~e & g);
      const t1 = (k + S1 + ch + K[i] + w[i]) | 0;
      const S0 = ((a >>> 2) | (a << 30)) ^ ((a >>> 13) | (a << 19)) ^ ((a >>> 22) | (a << 10));
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (S0 + maj) | 0;
      k = g; g = f; f = e; e = (d + t1) | 0; d = c; c = b; b = a; a = (t1 + t2) | 0;
    }
    h[0] += a; h[1] += b; h[2] += c; h[3] += d; h[4] += e; h[5] += f; h[6] += g; h[7] += k;
  }
  return Array.from(h, (v) => v.toString(16).padStart(8, '0')).join('');
}

// ---------------------------------------------------------------- chunks

/** The `font-chunk` frames for one file. The caller has already checked its size. */
export function chunkFont(input, { name, fileName }) {
  const bytes = toBytes(input);
  const sha256 = sha256Hex(bytes);
  const count = Math.max(1, Math.ceil(bytes.length / FONT_CHUNK_BYTES));
  const frames = [];
  for (let index = 0; index < count; index++) {
    const slice = bytes.subarray(index * FONT_CHUNK_BYTES, (index + 1) * FONT_CHUNK_BYTES);
    frames.push({ type: 'font-chunk', name, fileName, index, count, bytes: bytes.length, sha256, data: bytesToBase64(slice) });
  }
  return frames;
}

/**
 * One file from its chunks, in any order -- or the reason it cannot be.
 *
 * The checksum is the last word: a file that decodes, adds up to the right
 * length and still hashes wrong is refused, because handing `FontFace` a
 * truncated font fails later and less clearly than refusing it here.
 */
export function assembleFont(frames) {
  if (!Array.isArray(frames) || !frames.length) return { ok: false, reason: 'no chunks arrived' };
  const { count, bytes: total, sha256 } = frames[0];
  const parts = new Array(count);
  for (const f of frames) {
    if (f.count !== count || f.bytes !== total || f.sha256 !== sha256) {
      return { ok: false, reason: 'the chunks disagree about which file they belong to' };
    }
    const data = base64ToBytes(f.data);
    if (!data) return { ok: false, reason: `chunk ${f.index + 1} is not valid base64` };
    parts[f.index] = data;
  }
  for (let i = 0; i < count; i++) {
    if (!parts[i]) return { ok: false, reason: `chunk ${i + 1} of ${count} never arrived` };
  }
  const out = new Uint8Array(total);
  let at = 0;
  for (const p of parts) {
    if (at + p.length > total) return { ok: false, reason: 'the chunks add up to more than the file size' };
    out.set(p, at);
    at += p.length;
  }
  if (at !== total) return { ok: false, reason: `received ${at} of ${total} bytes` };
  if (sha256Hex(out) !== sha256) return { ok: false, reason: 'the file arrived damaged (checksum mismatch)' };
  return { ok: true, bytes: out };
}
```

```js
// ------------------------------------------------------------------ fonts

/**
 * Font files, found by what they ARE, sent when the app asks.
 *
 * Photoshop lists its fonts but never says where their files are, so the
 * plugin opens files in folders the user chose and reads each one's own
 * PostScript name (`fontNamesIn`). What it learns is kept in `fontIndex`, so
 * a file is opened once, not once per request. The index is only a hint: a
 * hit is re-read and re-checked before it is sent, so a stale entry costs a
 * read and never sends the wrong font.
 */
const FOLDERS_KEY = 'psdsync.fontFolders';
const INDEX_KEY = 'psdsync.fontIndex';
const FONT_FILE = /\.(ttf|otf|ttc)$/i;

let storage = null;
/** { token, label, folder } — `folder` is null when the token no longer resolves. */
let fontFolders = [];
/** "label|fileName" -> PostScript names inside that file. */
let fontIndex = {};
/** Requests are served one at a time; a request mid-scan waits its turn. */
let fontQueue = Promise.resolve();

function readJson(key, fallback) {
  try { const v = localStorage.getItem(key); return v ? JSON.parse(v) : fallback; } catch { return fallback; }
}
function writeJson(key, value) {
  try { localStorage.setItem(key, JSON.stringify(value)); } catch (e) { warn(`could not save ${key}: ${e.message}`); }
}
const saveFolders = () => writeJson(FOLDERS_KEY, fontFolders.map((f) => ({ token: f.token, label: f.label })));

async function restoreFontFolders() {
  fontIndex = readJson(INDEX_KEY, {});
  fontFolders = [];
  for (const saved of readJson(FOLDERS_KEY, [])) {
    let folder = null;
    try { folder = await storage.localFileSystem.getEntryForPersistentToken(saved.token); }
    catch (e) { warn(`fonts folder "${saved.label}" is no longer available: ${e && e.message ? e.message : e}`); }
    fontFolders.push({ token: saved.token, label: saved.label, folder });
  }
  if (fontFolders.length) info(`fonts folders: ${fontFolders.map((f) => f.label).join(', ')}`);
  renderFontFolders();
}

async function addFontFolder() {
  if (!storage) { fail("no require('uxp').storage — a folder cannot be chosen on this build"); return; }
  const folder = await storage.localFileSystem.getFolder();
  if (!folder) { info('no folder chosen'); return; }
  const label = folder.nativePath || folder.name;
  const token = await storage.localFileSystem.createPersistentToken(folder);
  fontFolders = fontFolders.filter((f) => f.label !== label);
  fontFolders.push({ token, label, folder });
  saveFolders();
  good(`fonts folder added: ${label}`);
  renderFontFolders();
}

function removeFontFolder(label) {
  fontFolders = fontFolders.filter((f) => f.label !== label);
  saveFolders();
  info(`fonts folder removed: ${label}`);
  renderFontFolders();
}

function renderFontFolders() {
  if (!ui.fontfolders) return;

/**
 * name -> {bytes, fileName} | {reason, detail}, reading as few files as it can.
 *
 * Files the index says hold a wanted name are read first; files it already
 * knows hold nothing wanted are skipped; everything else goes in
 * `candidateOrder`, and the scan stops once every name is answered.
 */
async function findFonts(names, folders) {
  const found = new Map();
  const remaining = new Set(names);
  for (const f of folders) {
    if (!remaining.size) break;
    let entries = [];
    try { entries = await f.folder.getEntries(); }
    catch (e) { warn(`could not list ${f.label}: ${e && e.message ? e.message : e}`); continue; }
    const files = entries.filter((e) => e.isFile && FONT_FILE.test(e.name));
    const byName = new Map(files.map((e) => [e.name, e]));
    const known = [];
    const unknown = [];
    for (const e of files) {
      const held = fontIndex[`${f.label}|${e.name}`];
      if (!held) unknown.push(e.name);
      else if (held.some((n) => remaining.has(n))) known.push(e.name);
    }
    for (const fileName of known.concat(candidateOrder([...remaining], unknown))) {
      if (!remaining.size) break;
      const key = `${f.label}|${fileName}`;
      let bytes;
      try { bytes = await byName.get(fileName).read({ format: storage.formats.binary }); }
      catch (e) {
        (fontIndex[key] || []).filter((n) => remaining.has(n)).forEach((n) => {
          remaining.delete(n);
          found.set(n, { reason: 'unreadable', detail: `${fileName}: ${e && e.message ? e.message : e}` });
        });
        continue;
      }
      const names = fontNamesIn(bytes);
      const held = !names || names.unsupported ? [] : names.collection ? names.fonts.map((x) => x.postscriptName) : [names.postscriptName];
      fontIndex[key] = held;
      for (const n of held) {
        if (!remaining.has(n)) continue;
        remaining.delete(n);
        found.set(n, names.collection
          ? { reason: 'collection', detail: `it is inside ${fileName}, a font collection, which cannot be sent yet` }
          : { bytes, fileName });
      }
    }
  }
  writeJson(INDEX_KEY, fontIndex);
  const where = `${folders.length} chosen folder${folders.length === 1 ? '' : 's'}`;
  for (const n of remaining) {
    found.set(n, { reason: 'not-found', detail: `not in the ${where}. An Adobe Fonts activation cannot be sent.` });
  }
  return found;
}

async function serveFonts(names) {
  info(`the app asked for ${names.length} font${names.length === 1 ? '' : 's'}: ${names.join(', ')}`);
  const usable = fontFolders.filter((f) => f.folder);
  if (!usable.length) {
    names.forEach((name) => send({ type: 'font-missing', name, reason: 'no-folder', detail: 'no fonts folder is chosen in the Photoshop plugin' }));
    warn('no fonts folder chosen — click "Choose fonts folder…" so fonts can be sent');
    return;
  }
  const started = Date.now();
  const found = await findFonts(names, usable);
  for (const name of names) {
    const hit = found.get(name);
    if (hit && hit.bytes) {
      if (hit.bytes.byteLength > FONT_MAX_BYTES) {
        send({ type: 'font-missing', name, reason: 'too-large', detail: `${hit.fileName} is ${Math.round(hit.bytes.byteLength / 1048576)} MB; the limit is 32 MB` });
        warn(`${name}: ${hit.fileName} is over 32 MB — not sent`);
        continue;
      }
      const frames = chunkFont(hit.bytes, { name, fileName: hit.fileName });
      frames.forEach(send);
      good(`sent ${name} from ${hit.fileName} (${Math.round(hit.bytes.byteLength / 1024)} KB, ${frames.length} chunk${frames.length === 1 ? '' : 's'})`);
    } else {
      const miss = hit || { reason: 'not-found', detail: '' };
      send({ type: 'font-missing', name, reason: miss.reason, detail: miss.detail || '' });
      warn(`${name}: ${miss.reason}${miss.detail ? ' — ' + miss.detail : ''}`);
    }
  }
  info(`font lookup took ${Date.now() - started} ms`);
}
```

```ts
import { Injectable, signal } from '@angular/core';

/**
 * Font files synced from Photoshop, kept on this device.
 *
 * The user asked for this so a second session does not send the same fonts
 * again. It is a CACHE: IndexedDB can be cleared by the OS under storage
 * pressure, and every failure here is a cache miss, never an exception --
 * the fonts simply get requested again.
 *
 * Metadata and bytes are separate stores, so listing what is cached (and
 * how big it is) never loads 150 MB of fonts into memory to answer.
 *
 * Only synced fonts come here. Hand-picked files still leave with the tab;
 * see the spec's Open section.
 */
export interface FontMeta {
  name: string;
  fileName: string;
  size: number;
  usedAt: number;
}

export interface FontCacheBackend {
  list(): Promise<FontMeta[]>;
  read(name: string): Promise<ArrayBuffer | null>;
  /** Writes the metadata, and the bytes when given. */
  write(meta: FontMeta, bytes?: ArrayBuffer): Promise<void>;
  remove(name: string): Promise<void>;
  clear(): Promise<void>;
}

export const FONT_CACHE_CAP_BYTES = 150 * 1024 * 1024;

@Injectable({ providedIn: 'root' })
export class FontCacheService {
  /** IndexedDB where there is one; memory otherwise (tests, and very old WebViews). */
  backend: FontCacheBackend = typeof indexedDB === 'undefined' ? memoryFontBackend() : indexedDbFontBackend();
  cap = FONT_CACHE_CAP_BYTES;
  now = () => Date.now();

  /** Bytes held. Refreshed on every write, and by `refresh()` when a screen opens. */
  readonly totalBytes = signal(0);

  async get(names: string[]): Promise<{ name: string; fileName: string; bytes: ArrayBuffer }[]> {
    try {
      const wanted = new Set(names);
      const out: { name: string; fileName: string; bytes: ArrayBuffer }[] = [];
      for (const meta of (await this.backend.list()).filter((m) => wanted.has(m.name))) {
        const bytes = await this.backend.read(meta.name);
        if (!bytes) continue;
        await this.backend.write({ ...meta, usedAt: this.now() });
        out.push({ name: meta.name, fileName: meta.fileName, bytes });
      }
      return out;
    } catch {
      return [];
    }
  }

  async put(name: string, fileName: string, bytes: ArrayBuffer): Promise<boolean> {
    if (bytes.byteLength > this.cap) return false;
    try {
      await this.backend.write({ name, fileName, size: bytes.byteLength, usedAt: this.now() }, bytes);
      await this.evict();
      return true;
    } catch {
      return false;
    }
  }

  async clear(): Promise<void> {
    try { await this.backend.clear(); } catch { /* nothing left to clear */ }
    this.totalBytes.set(0);
  }

  async refresh(): Promise<void> {
    try {
      const all = await this.backend.list();
      this.totalBytes.set(all.reduce((sum, m) => sum + m.size, 0));
    } catch {
      this.totalBytes.set(0);
    }
  }

  /** Least recently used first, until what is left fits. */
  private async evict(): Promise<void> {
    const all = (await this.backend.list()).sort((a, b) => a.usedAt - b.usedAt);
    let total = all.reduce((sum, m) => sum + m.size, 0);
    for (const meta of all) {
      if (total <= this.cap) break;
      await this.backend.remove(meta.name);
      total -= meta.size;
    }
    this.totalBytes.set(total);
  }
}

export function memoryFontBackend(): FontCacheBackend {
  const metas = new Map<string, FontMeta>();
  const blobs = new Map<string, ArrayBuffer>();
  return {
    list: async () => [...metas.values()].map((m) => ({ ...m })),
    read: async (name) => blobs.get(name) ?? null,
    write: async (meta, bytes) => { metas.set(meta.name, { ...meta }); if (bytes) blobs.set(meta.name, bytes); },
    remove: async (name) => { metas.delete(name); blobs.delete(name); },
    clear: async () => { metas.clear(); blobs.clear(); },
  };
}

function indexedDbFontBackend(): FontCacheBackend {
  let db: Promise<IDBDatabase> | null = null;
  const open = () => (db ??= new Promise<IDBDatabase>((resolve, reject) => {
    const req = indexedDB.open('psd-sync-fonts', 1);
    req.onupgradeneeded = () => {
      req.result.createObjectStore('meta', { keyPath: 'name' });
      req.result.createObjectStore('bytes');
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => { db = null; reject(req.error); };
  }));

  /** One transaction over both stores; resolves with `req`'s result once it commits. */
  const tx = <T>(mode: IDBTransactionMode, body: (t: IDBTransaction) => IDBRequest<T> | void): Promise<T | undefined> =>
    open().then((d) => new Promise<T | undefined>((resolve, reject) => {
      const t = d.transaction(['meta', 'bytes'], mode);
      const req = body(t);
      t.oncomplete = () => resolve(req ? req.result : undefined);
      t.onerror = () => reject(t.error);
      t.onabort = () => reject(t.error);
    }));

  return {
    list: async () => (await tx<FontMeta[]>('readonly', (t) => t.objectStore('meta').getAll())) ?? [],
    read: async (name) => (await tx<ArrayBuffer>('readonly', (t) => t.objectStore('bytes').get(name))) ?? null,
    write: async (meta, bytes) => {
      await tx('readwrite', (t) => {
        t.objectStore('meta').put(meta);
        if (bytes) t.objectStore('bytes').put(bytes, meta.name);
      });
    },
    remove: async (name) => { await tx('readwrite', (t) => { t.objectStore('meta').delete(name); t.objectStore('bytes').delete(name); }); },
    clear: async () => { await tx('readwrite', (t) => { t.objectStore('meta').clear(); t.objectStore('bytes').clear(); }); },
  };
}
```

```ts
  /**
   * After the tree: load what the device kept, ask Photoshop for the rest.
   * Only names the document is substituting are asked for, so a font the app
   * already draws is never sent again.
   */
  private async syncFonts(): Promise<void> {
    const missing = this.fonts.missingNames();
    if (!missing.length) return;
    const cached = await this.fonts.loadCached(missing);
    cached.forEach((name) => this.setFont(name, { state: 'cached', detail: 'Loaded from this device — synced earlier.' }));
    const rest = missing.filter((n) => !cached.includes(n)).slice(0, 64);
    if (!rest.length || !this.socket) return;
    rest.forEach((name) => this.setFont(name, { state: 'requested', detail: 'Asked Photoshop for this font…' }));
    this.send({ type: 'font-request', names: rest });
    this.armFontTimer();
  }

  private onFontChunk(frame: any): void {
    const list = this.assembling.get(frame.name) ?? [];
    list.push(frame);
    this.assembling.set(frame.name, list);
    this.setFont(frame.name, { state: 'receiving', detail: 'Receiving from Photoshop…', received: list.length, count: frame.count });
    this.armFontTimer();
    if (list.length < frame.count) return;

    this.assembling.delete(frame.name);
    const result = assembleFont(list);
    if (!result.ok) {
      this.setFont(frame.name, { state: 'failed', detail: `Not loaded: ${result.reason}.` });
      return;
    }
    void this.install(frame.name, frame.fileName, result.bytes);
  }

  private async install(name: string, fileName: string, bytes: Uint8Array): Promise<void> {
    await this.fonts.assignTo(name, [new File([bytes], fileName)]);
    if (this.fonts.missingNames().includes(name)) {
      this.setFont(name, { state: 'failed', detail: 'The file arrived, but this browser could not load it as a font.' });
      return;
    }
    this.setFont(name, { state: 'synced', detail: `Sent from Photoshop (${fileName}).` });
    await this.fontCache.put(name, fileName, bytes.buffer as ArrayBuffer);
  }

  private onFontMissing(frame: any): void {
    this.setFont(frame.name, { state: 'missing', detail: FONT_MISSING_COPY[frame.reason] ?? frame.detail });
  }

  private setFont(name: string, entry: FontSync): void {
    this.fontSync.update((all) => ({ ...all, [name]: entry }));
  }

  /** Re-armed by every chunk; fires only when the stream has gone quiet. */
  private armFontTimer(): void {
    this.clearFontTimer();
    this.fontTimer = setTimeout(() => this.expireFonts(), this.fontTimeoutMs);
  }

  private clearFontTimer(): void {
    if (this.fontTimer !== null) { clearTimeout(this.fontTimer); this.fontTimer = null; }
  }

  private expireFonts(): void {
    this.fontTimer = null;
    const all = this.fontSync();
    for (const [name, entry] of Object.entries(all)) {
      if (entry.state !== 'requested' && entry.state !== 'receiving') continue;
      this.assembling.delete(name);
      this.setFont(name, {
        state: 'failed',
        detail: entry.state === 'requested'
          ? 'Photoshop did not answer. Update the Photoshop plugin, then pair again.'
          : 'The transfer stopped part way. Pair again to retry.',
      });
    }
  }
```
